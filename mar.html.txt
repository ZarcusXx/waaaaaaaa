<script>

const canvas = document.getElementById("canvas");
const ctx = canvas.getContext("2d");

let width;
let height;

function resize() {

    const dpr = window.devicePixelRatio || 1;

    width = window.innerWidth;
    height = window.innerHeight;

    canvas.width = width * dpr;
    canvas.height = height * dpr;

    canvas.style.width = width + "px";
    canvas.style.height = height + "px";

    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
}

resize();

window.addEventListener("resize", resize);


// =====================================
// FUNCIÓN DEL CORAZÓN
// =====================================

function heart(u) {

    const x =
        16 * Math.pow(Math.sin(u), 3);

    const y =
        13 * Math.cos(u)
        - 5 * Math.cos(2 * u)
        - 2 * Math.cos(3 * u)
        - Math.cos(4 * u);

    return { x, y };
}


// =====================================
// CREAR RAYOS
// =====================================

const rays = [];

for (let i = 0; i < 140; i++) {

    const u =
        (i / 140) * Math.PI * 2
        + (Math.random() * 0.04 - 0.02);

    const h = heart(u);

    const length =
        9.8 + Math.random() * (11.2 - 9.8);

    rays.push({
        x: h.x * length,
        y: h.y * length
    });
}


// =====================================
// ANIMACIÓN
// =====================================

let step = 0;

function loop() {

    ctx.clearRect(0, 0, width, height);

    /*
        En Turtle el corazón estaba en el centro
        de una ventana de 600x600.

        Acá hacemos que se adapte
        automáticamente a la pantalla.
    */

    const scale =
        Math.min(width, height) / 600;

    const centerX = width / 2;
    const centerY = height / 2;


    const t =
        (step % 48) / 24.0;


    if (t <= 1.6) {

        const head =
            Math.min(1.0, t);

        const tail =
            Math.max(0.0, t - 0.35);


        const fade =
            t > 0.7
            ? Math.max(0, 1.0 - (t - 0.7))
            : 1.0;


        const red =
            Math.floor(170 * fade);


        // Color de los rayos
        ctx.strokeStyle =
            `rgb(${red}, 0, 26)`;

        ctx.lineWidth = 1;


        // =================================
        // DIBUJAR RAYOS
        // =================================

        for (const ray of rays) {

            const x1 =
                centerX +
                ray.x * tail * scale;

            const y1 =
                centerY -
                ray.y * tail * scale;


            const x2 =
                centerX +
                ray.x * head * scale;

            const y2 =
                centerY -
                ray.y * head * scale;


            ctx.beginPath();

            ctx.moveTo(x1, y1);
            ctx.lineTo(x2, y2);

            ctx.stroke();


            // =================================
            // PUNTO EN LA PUNTA
            // =================================

            if (t < 0.9) {

                ctx.fillStyle = "#ff1a40";

                ctx.beginPath();

                ctx.arc(
                    x2,
                    y2,
                    1.8 * scale,
                    0,
                    Math.PI * 2
                );

                ctx.fill();
            }
        }
    }


    step++;

    requestAnimationFrame(loop);
}


loop();

</script>
